Create a Supabase PostgreSQL schema with 5 tables for an AI news platform:

1. user_profiles:
   - id UUID PRIMARY KEY DEFAULT gen_random_uuid()

   - role TEXT NOT NULL CHECK (role IN ('student', 'professional', 'business', 'investor', 'citizen'))
   - goal TEXT
   - language TEXT DEFAULT 'english'
   - industry TEXT
   - ai_score INTEGER DEFAULT 0 CHECK (ai_score >= 0 AND ai_score <= 100)
   - created_at TIMESTAMPTZ DEFAULT NOW()

2. articles:
   - id UUID PRIMARY KEY DEFAULT gen_random_uuid()
   - title TEXT NOT NULL
   - url TEXT UNIQUE NOT NULL
   - source TEXT
   - category TEXT CHECK (category IN ('research', 'product', 'funding', 'regulation', 'opinion'))
   - original_text TEXT
   - eli5 TEXT
   - credibility INTEGER DEFAULT 50
   - published_at TIMESTAMPTZ
   - processed_at TIMESTAMPTZ DEFAULT NOW()
   - relevance_scores JSONB DEFAULT '{}'::jsonb

3. article_contexts:
   - id UUID PRIMARY KEY DEFAULT gen_random_uuid()
   - article_id UUID REFERENCES articles(id) ON DELETE CASCADE
   - user_role TEXT NOT NULL
   - why_it_matters TEXT
   - action_items JSONB DEFAULT '[]'::jsonb
   - language TEXT DEFAULT 'english'
   - localized_eli5 TEXT

4. user_interactions:
   - id UUID PRIMARY KEY DEFAULT gen_random_uuid()
   - user_id UUID REFERENCES user_profiles(id) ON DELETE CASCADE
   - article_id UUID REFERENCES articles(id) ON DELETE CASCADE
   - action_type TEXT CHECK (action_type IN ('read', 'action_click', 'share', 'bookmark'))

   - id UUID PRIMARY KEY DEFAULT gen_random_uuid()
